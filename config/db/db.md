create table public.users (
  id bigint generated by default as identity not null,
  username text not null,
  email text not null,
  password text not null,
  fname text null default 'NONE'::text,
  lname text null default 'NONE'::text,
  phone text null,
  is_active bigint not null default '1'::bigint,
  constraint users_pkey primary key (id),
  constraint users_email_key unique (email)
) TABLESPACE pg_default;

create table public.user_roles (
  user_role_id bigint generated by default as identity not null,
  user_id bigint null,
  role bigint null,
  constraint user_roles_pkey primary key (user_role_id),
  constraint user_roles_role_fkey foreign KEY (role) references roles (role_id) on update CASCADE on delete CASCADE,
  constraint user_roles_user_id_fkey foreign KEY (user_id) references users (id) on update CASCADE on delete CASCADE
) TABLESPACE pg_default;

create table public.tests (
  id bigint generated by default as identity not null,
  school_year_id bigint not null,
  subject_id bigint not null,
  teacher_id bigint not null,
  title text not null,
  time_limit bigint null,
  is_published bigint not null default '0'::bigint,
  created_at text not null default 'CURRENT_TIMESTAMP'::text,
  constraint tests_pkey primary key (id),
  constraint tests_school_year_id_fkey foreign KEY (school_year_id) references school_years (id),
  constraint tests_subject_id_fkey foreign KEY (subject_id) references subjects (id),
  constraint tests_teacher_id_fkey foreign KEY (teacher_id) references users (id)
) TABLESPACE pg_default;

create table public.teacher_subjects (
  id bigint generated by default as identity not null,
  teacher_id bigint not null,
  subject_id bigint not null,
  constraint teacher_subjects_pkey primary key (id),
  constraint teacher_subjects_subject_id_fkey foreign KEY (subject_id) references subjects (id) on update CASCADE on delete CASCADE,
  constraint teacher_subjects_teacher_id_fkey foreign KEY (teacher_id) references users (id) on update CASCADE on delete CASCADE
) TABLESPACE pg_default;

create table public.subjects (
  id bigint generated by default as identity not null,
  school_year_id bigint not null,
  subject_name text not null,
  subject_description text null,
  is_active bigint not null default '1'::bigint,
  constraint subjects_pkey primary key (id),
  constraint subjects_subject_name_key unique (subject_name),
  constraint subjects_school_year_id_fkey foreign KEY (school_year_id) references school_years (id)
) TABLESPACE pg_default;

create table public.students (
  id bigint generated by default as identity not null,
  school_year_id bigint not null,
  guardian_id bigint null,
  fname text not null,
  lname text not null,
  grade_level text null,
  constraint students_pkey primary key (id),
  constraint students_guardian_id_fkey foreign KEY (guardian_id) references guardians (id),
  constraint students_school_year_id_fkey foreign KEY (school_year_id) references school_years (id)
) TABLESPACE pg_default;

create table public.school_years (
  id bigint generated by default as identity not null,
  name text not null,
  start_date text not null,
  end_date text not null,
  is_active bigint not null default '0'::bigint,
  constraint school_years_pkey primary key (id)
) TABLESPACE pg_default;

create table public.roles (
  role_id bigint generated by default as identity not null,
  role text not null,
  constraint roles_pkey primary key (role_id)
) TABLESPACE pg_default;

create table public.questions (
  id bigint generated by default as identity not null,
  test_id bigint not null,
  question_text text not null,
  points text not null default '1'::text,
  constraint questions_pkey primary key (id),
  constraint questions_test_id_fkey foreign KEY (test_id) references tests (id) on update CASCADE on delete CASCADE
) TABLESPACE pg_default;

create table public.guardians (
  id bigint generated by default as identity not null,
  school_year_id bigint not null,
  fname text not null,
  lname text not null,
  phone text null,
  email text null,
  constraint guardians_pkey primary key (id),
  constraint guardians_school_year_id_fkey foreign KEY (school_year_id) references school_years (id)
) TABLESPACE pg_default;

create table public.choices (
  id bigint generated by default as identity not null,
  question_id bigint not null,
  choice_text text not null,
  is_correct bigint not null,
  constraint choices_pkey primary key (id),
  constraint choices_question_id_fkey foreign KEY (question_id) references questions (id) on update CASCADE on delete CASCADE
) TABLESPACE pg_default;
